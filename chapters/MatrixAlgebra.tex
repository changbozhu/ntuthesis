\section{向量变元的梯度}
向量变元$\vector{x}$的标量值函数$f:\mathbb{R}^{d}  \mapsto \mathbb{R}$就是把任意向量变元 $\vector{x} \in \mathbb{R}^{d}$ 映射到实数轴上唯一的点$y$，即 $y = f(\vector{x})$。接着，可以定义实值函数相对于（列）向量变元的梯度。
\begin{definition}[实值函数相对于列向量的梯度] \label{def:matAlg-scarlar-columnVectorGradient}
	实值函数$f(\vector{x})$相对于列向量$\vector{x} \in \bbR^{d}$的梯度是一个列向量，其第$i$个分量是函数$f$相对于列向量$\vector{x}$的第$i$个分量元素的偏导数，即
	$$
	\nabla_{\vector{x}} f(\vector{x}) = \frac{\partial f(\vector{x})}{ \partial \vector{x} } = \left[ \frac{\partial f}{\partial x_1}, \frac{\partial f}{\partial x_2}, \cdots, \frac{\partial f}{\partial x_d} \right]^{\top}
	$$
\end{definition}
然后，实值函数相对于行向量的梯度可如下定义，
\begin{definition}[实值函数相对于行向量的梯度]  \label{def:matAlg-scarlar-rowVectorGradient}
	实值函数$f(\vector{x})$相对于行向量$\vector{x}^{\top} \in \bbR^{d}$的梯度是一个行向量，其第$i$个分量是函数$f$相对于行向量$\vector{x}^{\top}$的第$i$个分量元素的偏导数，即
	$$
	\nabla_{\vector{x}^{\top}} f(\vector{x}) = \frac{\partial f(\vector{x})}{ \partial \vector{x}^{\top} } = \left[ \frac{\partial f}{\partial x_{1}}, \frac{\partial f}{\partial x_{2}}, \cdots, \frac{\partial f}{\partial x_{d}} \right]
	$$
\end{definition}
向量变元$\vector{x} \in \mathbb{R}^d$的梯度负向称为\emph{梯度流}，记为$ \dot{\vector{x}} = -  \frac{\partial f(\vector{x})}{ \partial \vector{x} }$。若函数$f_i(\vector{x}), 1 \le i \le m$是向量变元$\vector{x} \in \mathbb{R}^d$的实值函数，那么函数$\vector{f}(\vector{x})=[f_1(\vector{x}), f_2(\vector{x}), \ldots, f_m(\vector{x})]^{\top}$是向量变元$\vector{x}$的向量值函数。进一步地，基于定义\ref{def:matAlg-scarlar-columnVectorGradient}及定义\ref{def:matAlg-scarlar-rowVectorGradient}，向量值函数$\vector{f}(\vector{x})$相对于向量变元$\vector{x}$的梯度可由下式定义
\begin{equation*}
\begin{split}
\frac{\partial \vector{f}^{\top} (\vector{x}) }{ \partial \vector{x} }
=& \nabla_{\vector{x}} \vector{f}(\vector{x})=  \left[ \frac{\partial f^{(1)}}{\partial  \vector{x}}, \frac{\partial  f^{(2)}}{\partial  \vector{x}}, \ldots, \frac{\partial f^{(m)}}{\partial \vector{x}} \right] \\
=&\begin{bmatrix}
	&\frac{\partial f_1}{\partial x_1}  &\frac{\partial f_2}{\partial x_1}  &\cdots  &\frac{\partial f_n}{\partial x_1} \\
	&\frac{\partial f_1}{\partial x_2}  &\frac{\partial f_2}{\partial x_2}  &\cdots  &\frac{\partial f_n}{\partial x_2} \\
	&\vdots 							&\vdots								&		 &\vdots							\\
	&\frac{\partial f_1}{\partial x_m}  &\frac{\partial f_2}{\partial x_m}  &\cdots  &\frac{\partial f_n}{\partial x_m}
\end{bmatrix}
\in \mathbb{R}^{m \times n}
\end{split}
\end{equation*}
全微分为
$ \mathrm{d} \vector{f} = (\frac{\partial \vector{f}^{\top}}{\partial \vector{x}})^{\top}   \mathrm{d}\vector{x} $。

实值函数$f(\vector{x})$相对于向量$\vector{x}$的梯度满足下列性质，
\begin{enumerate}	
	\item 若$f(\vector{x}) = c$为常数，那么$\nabla_{\vector{x}} f(\vector{x}) = \vector{0}$。
	\item 线性法则：若$f(\vector{x})$和$g(\vector{x})$均是$\vector{x}$的实值函数，$c_1, c_2$为常数，那么
	      $$\frac{\partial [ c_1 f(\vector{x}) + c_2 g(\vector{x}) ] }{\partial \vector{x}} = c_1 \frac{ \partial f(\vector{x})}{\partial \vector{x}}  + c_2 \frac{ \partial g(\vector{x})}{\partial \vector{x}} $$
	\item 乘法规则：若$f(\vector{x})$和$g(\vector{x})$均是$\vector{x}$的实值函数，那么
	      $$\frac{\partial [ f(\vector{x}) g(\vector{x}) ] }{\partial \vector{x}} = f(\vector{x}) \frac{ \partial g(\vector{x})}{\partial \vector{x}}  + g(\vector{x}) \frac{ \partial f(\vector{x})}{\partial \vector{x}} $$
	\item 商式法则：若$f(\vector{x})$和$g(\vector{x}) \neq 0 $均是$\vector{x}$的实值函数，那么
	      $$\frac{\partial [ \frac{f(\vector{x})}{g(\vector{x})} ] }{\partial \vector{x}} = \frac{ g(\vector{x}) \frac{ \partial f(\vector{x})}{\partial \vector{x}}  - f(\vector{x}) \frac{ \partial g(\vector{x})}{\partial \vector{x}} }{g(\vector{x})^2} $$
	\item 链式法则：若$f(\vector{y})$是向量$\vector{y}$的实值函数, 并且$\vector{y}$是向量变元$\vector{x}$的向量值函数，那么
	      $$\frac{\partial f }{\partial \vector{x}} = \frac{\partial \vector{y}^{\top} }{\partial \vector{x}}  \frac{\partial f(\vector{y}) }{\partial \vector{y}}  $$
	\item 若向量$\vector{a}$与$\vector{x}$是无关的向量，那么
	      $$\frac{\partial \vector{a}^{\top} \vector{x}}{\partial \vector{x}} = \frac{\partial \vector{x}^{\top} \vector{a}}{\partial \vector{x}} = \vector{a} $$
	\item $\frac{\partial \vector{x}^{\top}}{ \partial \vector{x}} = \Matrix{I}$是单位矩阵。
	\item 线性映射的梯度：若$\Matrix{W},\vector{b}$分别是矩阵和常向量，$\vector{x}$是向量变元，
		  则线性映射$\vector{y} = \Matrix{W} \vector{x} + \vector{b}$的梯度矩阵为
		  $$\frac{\partial \vector{y}^{\top}}{\partial \vector{x}} = \Matrix{W}$$
	\item 若向量$\vector{a}$与$\vector{x}$是相关的向量，且$\vector{y}$是向量$\vector{x}$的向量值函数，那么
	      $$\frac{\partial \vector{a}^{\top} \vector{y}}{\partial \vector{x}} = \frac{\partial \vector{y}^{\top} \vector{a}}{\partial \vector{x}} = \frac{\partial \vector{y}^{\top}}{\partial \vector{x}} \vector{a} $$
	\item 若矩阵$\Matrix{A}$和向量$\vector{y}$均与向量$\vector{x}$是无关的，那么
	      $$\frac{\partial \vector{x}^{\top} \Matrix{A} \vector{y} }{\partial \vector{x}} = \Matrix{A} \vector{y}, \frac{\partial \vector{y}^{\top} \Matrix{A} \vector{x} }{\partial \vector{x}} = \Matrix{A}^{\top} \vector{y} $$
	\item 若矩阵$\Matrix{A}$与向量$\vector{x}$是无关的，那么
	      $$\frac{\partial \vector{x}^{\top} \Matrix{A} \vector{x} }{\partial \vector{x}} = (\Matrix{A} + \Matrix{A}^{\top}) \vector{x}$$
	\item 若矩阵$\Matrix{A}$与向量$\vector{x}$是无关的，而向量$\vector{y}$是向量$\vector{x}$的向量值函数，那么
	      $$\frac{\partial \vector{y}^{\top} \Matrix{A} \vector{y} }{\partial \vector{x}} = \frac{\partial \vector{y}^{\top}}{\partial \vector{x}}(\Matrix{A} + \Matrix{A}^{\top}) \vector{y}$$
	\item 若矩阵$\Matrix{A}$与向量$\vector{x}$是无关的，而向量$\vector{y},\vector{z}$均是向量$\vector{x}$的向量值函数，那么
	      $$\frac{\partial \vector{y}^{\top} \Matrix{A} \vector{z} }{\partial \vector{x}} = \frac{\partial \vector{y}^{\top}}{\partial \vector{x}}\Matrix{A} \vector{y}  +  \frac{\partial \vector{z}^{\top}}{\partial \vector{x}}\Matrix{A}^{\top} \vector{z}$$
\end{enumerate}
